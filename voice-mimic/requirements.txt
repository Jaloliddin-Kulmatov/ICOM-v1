# VoiceMimic — real voice cloning stack (verified on macOS arm64, Python 3.12).
#
# Easiest path: run ./setup.sh (creates a venv OUTSIDE iCloud and installs all
# of this in the right order). These pins matter — see the comments.

Flask>=3.0

# Real zero-shot voice cloning (XTTS-v2). coqui-tts does NOT pull torch itself.
coqui-tts>=0.24

# transformers 5.x removed symbols XTTS still imports (isin_mps_friendly), so
# XTTS only works on the 4.57 line.
transformers>=4.57,<5

# torch 2.9+ requires the torchcodec audio backend, which needs system FFmpeg.
# 2.8.0 uses soundfile/torchaudio instead — no FFmpeg needed.
torch==2.8.0
torchaudio==2.8.0
